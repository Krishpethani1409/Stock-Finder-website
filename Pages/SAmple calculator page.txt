  const [monthlyInvestment, setMonthlyInvestment] = useState(5000);
  const [durationYears, setDurationYears] = useState(5);
  const [expectedRate, setExpectedRate] = useState(12);
  const [frequency, setFrequency] = useState('MONTHLY'); // 'MONTHLY' | 'QUARTERLY' | 'YEARLY'

  // Accordion State for FAQs
  const [openFaq, setOpenFaq] = useState(null);

  // ── DYNAMIC SIP MATHEMATICAL ENGINE ──
  const sipResult = useMemo(() => {
    const P = Math.max(100, Number(monthlyInvestment) || 0);
    const r = Math.max(1, Math.min(30, Number(expectedRate) || 12));
    const t = Math.max(1, Math.min(30, Number(durationYears) || 5));

    // Frequency multiplier adjustments
    let nPeriods = t * 12;
    let ratePerPeriod = r / 12 / 100;

    if (frequency === 'QUARTERLY') {
      nPeriods = t * 4;
      ratePerPeriod = r / 4 / 100;
    } else if (frequency === 'YEARLY') {
      nPeriods = t;
      ratePerPeriod = r / 100;
    }

    const totalInvested = P * nPeriods;
    // Formula: FV = P * [((1 + i)^n - 1) / i] * (1 + i)
    const futureValue = P * ((Math.pow(1 + ratePerPeriod, nPeriods) - 1) / ratePerPeriod) * (1 + ratePerPeriod);
    const estimatedReturns = Math.max(0, futureValue - totalInvested);

    // Amortization Schedule (Year-by-Year Table)
    const schedule = [];
    for (let year = 1; year <= t; year++) {
      const periodsSoFar = frequency === 'QUARTERLY' ? year * 4 : frequency === 'YEARLY' ? year : year * 12;
      const yrInvested = P * periodsSoFar;
      const yrFV = P * ((Math.pow(1 + ratePerPeriod, periodsSoFar) - 1) / ratePerPeriod) * (1 + ratePerPeriod);
      const yrReturns = yrFV - yrInvested;

      schedule.push({
        year,
        invested: Math.round(yrInvested),
        returns: Math.round(yrReturns),
        total: Math.round(yrFV)
      });
    }

    return {
      invested: Math.round(totalInvested),
      returns: Math.round(estimatedReturns),
      total: Math.round(futureValue),
      schedule
    };
  }, [monthlyInvestment, durationYears, expectedRate, frequency]);

  // Donut Chart SVG Calculation
  const donutData = useMemo(() => {
    const total = sipResult.total || 1;
    const investedPct = Math.min(100, Math.max(0, (sipResult.invested / total) * 100));
    const returnsPct = 100 - investedPct;
    
    // Donut Stroke Circumference (r=40, C = 2 * PI * 40 = ~251.32)
    const C_LEN = 251.32;
    const investedDash = (investedPct / 100) * C_LEN;
    const returnsDash = C_LEN - investedDash;

    return {
      investedPct: investedPct.toFixed(1),
      returnsPct: returnsPct.toFixed(1),
      investedDash,
      returnsDash,
      offset: -investedDash
    };
